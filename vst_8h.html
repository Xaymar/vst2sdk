uint32_t expand_vst_version(uint32_t v) {
if (v < 10) { //
return v << 24;
}
uint8_t major = v / 1000;
uint8_t minor = (v / 100) % 10;
uint8_t revision = (v / 10) % 10;
uint8_t patch = v % 10;
return (major << 24) | (minor << 16) | (revision << 8) | patch;
}
